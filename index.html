import pyautogui import time import keyboard from colorama import init, Fore, Style init(autoreset=True) recoil = False rapid = False sens = False def banner(): print(Fore.CYAN + Style.BRIGHT) print("╔═══════════╗") print("║ FF PRO TOOLKIT v2.0 ║") print("║ Solo uso educativo / Salas ║") print("╚═══════════╝") print(Style.RESET_ALL) def menu(): print(Fore.YELLOW + "[F1]" + Fore.WHITE + " = Control de Retroceso [ON/OFF]") print(Fore.YELLOW + "[F2]" + Fore.WHITE + " = Macro Disparo Rápido [ON/OFF]") print(Fore.YELLOW + "[F3]" + Fore.WHITE + " = Sensibilidad Pro [ON/OFF]") print(Fore.YELLOW + "[F9]" + Fore.WHITE + " = Salir del programa") print(Fore.CYAN + "\n[*] Esperando comando..." + Style.RESET_ALL) def toggle_recoil(): global recoil recoil = not recoil print(Fore.GREEN + f"[+] Retroceso: {'ACTIVADO' if recoil else 'DESACTIVADO'}") def toggle_rapid(): global rapid rapid = not rapid print(Fore.GREEN + f"[+] Disparo Rápido: {'ACTIVADO' if rapid else 'DESACTIVADO'}") def toggle_sens(): global sens sens = not sens print(Fore.GREEN + f"[+] Sens Pro: {'ACTIVADO' if sens else 'DESACTIVADO'}") banner() menu() keyboard.add_hotkey('f1', toggle_recoil) keyboard.add_hotkey('f2', toggle_rapid) keyboard.add_hotkey('f3', toggle_sens) while True: if recoil and keyboard.is_pressed('left mouse'): pyautogui.moveRel(0, 2) time.sleep(0.007) if rapid and keyboard.is_pressed('left mouse'): pyautogui.click() time.sleep(0.025) if keyboard.is_pressed('f9'): print(Fore.RED + "[!] Saliendo...") break time.sleep(0.001)
